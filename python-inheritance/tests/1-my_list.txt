#!/usr/bin/python3
"""
MyList klası üçün unittest ssenariləri.
"""
import unittest
MyList = __import__('1-my_list').MyList


class TestMyList(unittest.TestCase):
    """MyList klasının metodlarını yoxlayan testlər."""

    def test_inheritance(self):
        """MyList-in list-dən miras aldığını yoxlayır."""
        my_list = MyList()
        self.assertIsInstance(my_list, list)

    def test_print_sorted_output(self):
        """print_sorted metodunun düzgün sıralama ilə çap etdiyini yoxlayır."""
        import io
        from contextlib import redirect_stdout

        my_list = MyList([5, 2, 4, 1, 3])
        f = io.StringIO()
        with redirect_stdout(f):
            my_list.print_sorted()
        self.assertEqual(f.getvalue(), "[1, 2, 3, 4, 5]\n")

    def test_original_list_unchanged(self):
        """Sıralamadan sonra orijinal siyahının dəyişmədiyini yoxlayır."""
        my_list = MyList([3, 1, 2])
        my_list.print_sorted()
        self.assertEqual(my_list, [3, 1, 2])

    def test_empty_list(self):
        """Boş siyahı ilə davranışı yoxlayır."""
        my_list = MyList()
        import io
        from contextlib import redirect_stdout
        f = io.StringIO()
        with redirect_stdout(f):
            my_list.print_sorted()
        self.assertEqual(f.getvalue(), "[]\n")

    def test_strings_raises_error(self):
        """Siyahıda həm int, həm string olarsa xəta verməsini yoxlayır."""
        my_list = MyList([1, "hello"])
        with self.assertRaises(TypeError):
            my_list.print_sorted()

if __name__ == "__main__":
    unittest.main()
